{{/* 收集文章容器的数据属性：小体大写、对齐、类型、布局、段首缩排 上下文：页面 返回：dict{ smallCaps, align, type, layout, indent } Usage: {{ partial "utils/data-attributes.html" . }} */}} {{- $smallCaps := .Params.smallCaps | default .Site.Params.enableSmallCaps | default false -}} {{- $align := .Params.align | default "default" -}} {{- if ne .Type "poetry" -}} {{- if and $.Site.Params.enableJustify (eq $align "default") -}} {{- $align = "justify" -}} {{- end -}} {{- end -}} {{- $type := .Type -}} {{- $layout := .Params.layout -}} {{- $indent := false -}} {{- if and .Site.Params.enableParagraphIndent (ne .Type "poetry") -}} {{- if .Params.indent | default (eq .Site.Params.paragraphStyle "indent") -}} {{- $indent = true -}} {{- end -}} {{- end -}} {{- return dict "smallCaps" $smallCaps "align" $align "type" $type "layout" $layout "indent" $indent -}}
