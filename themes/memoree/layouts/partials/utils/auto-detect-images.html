{{/* 从正文 中提取图片 URL（utils/images.html 的辅助函数） 上下文：页面 返回：URL 切片 */}} {{- $imgs := slice -}} {{- if and .IsPage (in .Site.Params.mainSections .Section) -}} {{- range (findRE `]*?src="/?([^"]+)` .Content | uniq) -}} {{- $url := replaceRE `]*?src="/?([^"]+)` `$1` . -}} {{- $imgs = union $imgs (slice $url) -}} {{- end -}} {{- end -}} {{- return $imgs -}}
