{{/* 从正文 中提取图片 URL（utils/images.html 的辅助函数） 上下文：页面 返回：URL 切片 */}} {{- $imgs := slice -}} {{- $host := .Site.Params.imageHostURL | default "" -}} {{- if and .IsPage (in .Site.Params.mainSections .Section) -}} {{- range (findRE `]*?(src|data-src)="[^"]+"` .Content | uniq) -}} {{- if not (strings.Contains . "post-img-lqip") -}} {{- $url := "" -}} {{- $match := findRE `data-src="([^"]+)"` . 1 -}} {{- if $match -}} {{- $url = replaceRE `data-src="([^"]+)"` `$1` (index $match 0) -}} {{- else -}} {{- $srcMatch := findRE `src="([^"]+)"` . 1 -}} {{- with $srcMatch -}} {{- $url = replaceRE `src="([^"]+)"` `$1` (index . 0) -}} {{- end -}} {{- end -}} {{- /* 只收站内相对路径与图床 URL；外链图片（如友链头像）不进 OG/JSON-LD */ -}} {{- if or (not (hasPrefix $url "http")) (and $host (hasPrefix $url $host)) -}} {{- $imgs = union $imgs (slice $url) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $imgs -}}
